# 개인 청약 프로필 스키마

`profile.yaml`(프로젝트 루트)의 필드 정의. 각자 처음 한 번 세팅한다. 모르는 값은 비워두면(`null`)
매칭에서 "해당 조건 판단 보류"로 처리하고, 리포트 notes에 "프로필 미입력: {필드}"로 안내한다.

```yaml
# ── 주택 보유 ──
household_status: 무주택세대주        # 무주택세대주 | 무주택세대원 | 1주택 | 다주택
homeless_period_years: 5             # 무주택 기간(년). 가점 32점 만점=15년+. 모르면 null

# ── 세대 구성 (특별공급 자격 판단) ──  복수 선택
household_types:
  - 신혼부부                          # 혼인신고 7년 이내
  # - 예비신혼                        # 혼인신고 전(공공 신혼특공만)
  # - 신생아                          # 공고일 기준 2년 이내 출생 자녀
  # - 미성년자녀
  # - 다자녀                          # 미성년 자녀 2명 이상 (2023.3 완화)
  # - 노부모부양                      # 만65세↑ 직계존속 3년 이상 부양, 세대주만
  # - 청년                            # 만19~39세 미혼 무주택 (공공주택 특공만)
  # - 1인가구                         # 미혼 단독
  # - 생애최초                        # 세대 전원 과거 주택소유 이력 없음 + 소득세 5년↑ 납부
dependents_count: 2                   # 부양가족 수(본인 제외 or 세대원 기준). 가점 35점 만점=7인+
income_tax_years: null                # (생애최초용) 소득세 납부 연수. 5 미만이면 생애최초 특공 제외
won_within_5y: false                  # 최근 5년 내 본인·세대원 청약 당첨 이력 → 규제지역 1순위 강등

# ── 청약통장 ──
account_type: 종합저축                # 종합저축 | 청약예금 | 청약부금(85㎡ 이하만) | 없음
account_join_years: 6                 # 가입 기간(년). 가점 17점 만점=15년+. 1순위: 비규제 1년·규제 2년
account_payment_count: 72             # 납입 횟수(회). 국민주택 1순위: 비규제 12회·규제 24회
account_total_manwon: 1200            # 납입 인정액 총액(만원). 국민주택 1순위 저축총액
account_monthly_manwon: 10            # 월 납입액(만원): 10 또는 25(2026 인정한도 상향)
spouse_account_join_years: null       # (선택) 배우자 통장 가입기간(년). 민영 가점 최대 +3점 합산(§8-4)

# ── 비아파트 보유 (무주택 인정 판단, §8-2) ──
non_apt_property: null                # (선택) 빌라/오피스텔 보유 시: {area_m2, gongsi_manwon}
                                      # 전용 85㎡↓ & 공시가 수도권 5억(지방 3억)↓ 이면 무주택 인정

# ── 소득 (특공 우선/일반/추첨 단계 판단) ──
monthly_income_manwon: 750            # 세대 월평균소득(만원, 세전). 2025 도시근로자 3인↓ 100% = 753만
household_size: 3                     # 가구원 수: 3(이하) | 4 | 5  → 소득 기준표 선택
dual_income: true                     # 맞벌이 여부(부부 모두 소득) → 상한 완화(예: 신혼 100→120%)
child_after_2023: false               # 2023.3.28 이후 출생(임신·입양 포함) 자녀 → 소득·자산 기준 +20%p
income_level: 기준소득이하            # 월소득 미입력 시 근사용: 기준소득이하 | 기준소득초과 | 모름
# 세대유형 추가 옵션: 자녀6세이하(공공 신혼특공 대상), 한부모(6세↓ 자녀·공공 신혼특공)

# ── 자금·거주지 ──
residence_area: 서울                  # 신청자 주민등록 거주지: 서울 | 광역시(인천 등) | 기타(경기 등). 민영 예치금 기준표 선택용
deposit_capacity_manwon: 1500         # 통장 예치금(만원). 민영 1순위: 거주지×면적 기준표(서울 85㎡↓300…모든면적 1,500)

# ── 관심 조건 ──
regions:                              # 관심 수도권 세부지역(문자열 부분매칭). 비우면 수도권 전체
  - 서울
  - 경기 성남
  - 인천
area_pref:                            # 선호 전용면적(㎡). 비우면 전체
  - 59
  - 84
purpose: 실거주                       # 실거주 | 시세차익 | 둘다

# ── 전략 성향 ──
strategy_aggressive: true             # 빈집털이(애매평형/타워형/뒷알파벳/소수세대) 신호를 적극 노출할지
```

## 파생 계산 (매칭 엔진이 자동)
- **예상 가점** = 무주택기간점수 + 부양가족점수 + 통장기간점수 (지식베이스 §5 표).
- **국민 vs 민영 성향**: `account_monthly_manwon>=25` 또는 `account_total_manwon>=2000` → 국민주택 1순위 후보. 아니면 민영 추첨/예치 전략.
- **특공 자격 후보**: `household_types` × 지식베이스 §3 매핑.
- **추첨 우위 여부**: 예상 가점 < 55 → 민영은 추첨제 중심 안내.

## 입력 편의
- 프로필이 없으면(`profile.yaml` 부재) 매칭 단계를 건너뛰고 일반 리포트만 생성하며, 리포트에
  "개인 맞춤을 원하면 profile.yaml을 세팅하세요" 안내를 넣는다.
- 초기 세팅은 `profile.example.yaml`을 복사해 값만 바꾸면 된다.
